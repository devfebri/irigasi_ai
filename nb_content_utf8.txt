import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.preprocessing import StandardScaler
import warnings
warnings.filterwarnings('ignore')

pd.set_option('display.max_columns', None)
plt.style.use('seaborn-v0_8-whitegrid')

# Load data bersih
df = pd.read_csv('dataset/data_cleaned.csv', parse_dates=['waktu_pembacaan'])
df = df.sort_values('waktu_pembacaan').reset_index(drop=True)

print('=== ≡ƒôé Data Bersih Dimuat ===')
print(f'Baris  : {len(df):,}')
print(f'Kolom  : {df.shape[1]}')
print(df.head(3))

--- CELL 2 (code) ---
# ============================================================
# CELL 2: Target Variable ΓÇô Rata-rata Kelembaban Tanah 3 Node
# ============================================================
# Target utama: rata-rata kelembaban tanah dari 3 node
df['kelembaban_tanah_avg'] = df[['kelembaban_tanah_1',
                                  'kelembaban_tanah_2',
                                  'kelembaban_tanah_3']].mean(axis=1)

# Target prediksi: nilai BERIKUTNYA (t+1) ΓåÆ time-series forecasting
df['target_kelembaban_next'] = df['kelembaban_tanah_avg'].shift(-1)

print('=== ≡ƒÄ» Target Variable ===')
print(f'kelembaban_tanah_avg (saat ini)  : mean={df["kelembaban_tanah_avg"].mean():.2f}%')
print(f'target_kelembaban_next (t+1)     : mean={df["target_kelembaban_next"].mean():.2f}%')
print()
print(df[['waktu_pembacaan', 'kelembaban_tanah_avg', 'target_kelembaban_next']].head(5))

--- CELL 3 (code) ---
# ============================================================
# CELL 3: Feature Engineering ΓÇô Fitur Waktu
# ============================================================
print('=== ΓÅ░ Membuat Fitur Waktu ===')

df['jam']         = df['waktu_pembacaan'].dt.hour
df['menit']       = df['waktu_pembacaan'].dt.minute
df['hari']        = df['waktu_pembacaan'].dt.day
df['bulan']       = df['waktu_pembacaan'].dt.month
df['hari_minggu'] = df['waktu_pembacaan'].dt.dayofweek  # 0=Senin, 6=Minggu
df['is_weekend']  = (df['hari_minggu'] >= 5).astype(int)

# Fitur siklus sinus/cosinus untuk jam (menghindari diskontinuitas 23ΓåÆ0)
df['jam_sin'] = np.sin(2 * np.pi * df['jam'] / 24)
df['jam_cos'] = np.cos(2 * np.pi * df['jam'] / 24)

# Periode hari: pagi, siang, sore, malam
df['periode_hari'] = pd.cut(df['jam'],
                             bins=[-1, 5, 11, 17, 23],
                             labels=[0, 1, 2, 3])  # 0=malam, 1=pagi, 2=siang, 3=sore
df['periode_hari'] = df['periode_hari'].astype(int)

print('Fitur waktu dibuat: jam, menit, hari, bulan, hari_minggu, is_weekend, jam_sin, jam_cos, periode_hari')

--- CELL 4 (code) ---
# ============================================================
# CELL 4: Feature Engineering ΓÇô Lag Features
# ============================================================
print('=== ≡ƒôë Membuat Lag Features ===')

# Lag pada kelembaban tanah rata-rata
lag_steps = [1, 5, 10, 30, 60]
for lag in lag_steps:
    df[f'kt_avg_lag_{lag}'] = df['kelembaban_tanah_avg'].shift(lag)
    print(f'  kt_avg_lag_{lag}: kelembaban rata-rata {lag} langkah sebelumnya')

# Lag pada masing-masing node (hanya lag 1 dan lag 5 untuk efisiensi)
for node in [1, 2, 3]:
    col = f'kelembaban_tanah_{node}'
    df[f'kt{node}_lag_1'] = df[col].shift(1)
    df[f'kt{node}_lag_5'] = df[col].shift(5)

# Lag curah hujan
df['hujan_lag_1']  = df['curah_hujan'].shift(1)
df['hujan_lag_5']  = df['curah_hujan'].shift(5)
df['hujan_lag_30'] = df['curah_hujan'].shift(30)

print('\nLag features selesai dibuat.')

--- CELL 5 (code) ---
# ============================================================
# CELL 5: Feature Engineering ΓÇô Rolling Statistics
# ============================================================
print('=== ≡ƒôè Membuat Rolling Statistics ===')

# Interval 1 menit ΓåÆ 1 jam = 60 langkah, 6 jam = 360 langkah
windows = {'1jam': 60, '6jam': 360}

for window_name, window_size in windows.items():
    # Rolling mean kelembaban tanah
    df[f'kt_avg_roll_mean_{window_name}'] = df['kelembaban_tanah_avg'].shift(1).rolling(window=window_size, min_periods=10).mean()
    df[f'kt_avg_roll_std_{window_name}']  = df['kelembaban_tanah_avg'].shift(1).rolling(window=window_size, min_periods=10).std()
    df[f'kt_avg_roll_min_{window_name}']  = df['kelembaban_tanah_avg'].shift(1).rolling(window=window_size, min_periods=10).min()
    df[f'kt_avg_roll_max_{window_name}']  = df['kelembaban_tanah_avg'].shift(1).rolling(window=window_size, min_periods=10).max()
    
    # Rolling mean curah hujan
    df[f'hujan_roll_sum_{window_name}'] = df['curah_hujan'].shift(1).rolling(window=window_size, min_periods=1).sum()
    
    print(f'  Rolling {window_name} ({window_size} langkah): mean, std, min, max KT + sum hujan')

# Tren: selisih kelembaban dari t-1 ke sekarang
df['kt_avg_diff_1']  = df['kelembaban_tanah_avg'].diff(1)
df['kt_avg_diff_5']  = df['kelembaban_tanah_avg'].diff(5)
df['kt_avg_diff_30'] = df['kelembaban_tanah_avg'].diff(30)

print('\nRolling features selesai dibuat.')

--- CELL 6 (code) ---
# ============================================================
# CELL 6: Feature Engineering ΓÇô Fitur Interaksi & Gabungan
# ============================================================
print('=== ≡ƒöù Membuat Fitur Interaksi ===')

# Rata-rata kelembaban udara 3 node
df['ku_avg'] = df[['kelembaban_udara_1', 'kelembaban_udara_2', 'kelembaban_udara_3']].mean(axis=1)

# Rata-rata suhu udara 3 node
df['suhu_avg'] = df[['suhu_udara_1', 'suhu_udara_2', 'suhu_udara_3']].mean(axis=1)

# Rata-rata tekanan udara 3 node
df['tekanan_avg'] = df[['tekanan_udara_1', 'tekanan_udara_2', 'tekanan_udara_3']].mean(axis=1)

# Vapor Pressure Deficit (VPD) ΓÇô indikator evapotranspirasi
# VPD Γëê (1 - RH/100) * 0.6108 * exp(17.27 * T / (T + 237.3))
df['vpd_1'] = (1 - df['kelembaban_udara_1']/100) * 0.6108 * np.exp(17.27 * df['suhu_udara_1'] / (df['suhu_udara_1'] + 237.3))
df['vpd_2'] = (1 - df['kelembaban_udara_2']/100) * 0.6108 * np.exp(17.27 * df['suhu_udara_2'] / (df['suhu_udara_2'] + 237.3))
df['vpd_3'] = (1 - df['kelembaban_udara_3']/100) * 0.6108 * np.exp(17.27 * df['suhu_udara_3'] / (df['suhu_udara_3'] + 237.3))
df['vpd_avg'] = df[['vpd_1', 'vpd_2', 'vpd_3']].mean(axis=1)

# Perbedaan antar node (deteksi spatial variability)
df['kt_diff_node12'] = df['kelembaban_tanah_1'] - df['kelembaban_tanah_2']
df['kt_diff_node13'] = df['kelembaban_tanah_1'] - df['kelembaban_tanah_3']
df['kt_diff_node23'] = df['kelembaban_tanah_2'] - df['kelembaban_tanah_3']

print('Fitur interaksi: ku_avg, suhu_avg, tekanan_avg, vpd_1/2/3/avg, kt_diff_node12/13/23')

df = df.dropna(subset=['target_kelembaban_next'])

# ============================================================
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import json
import pickle
import os
import warnings
warnings.filterwarnings('ignore')

from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score
from sklearn.inspection import permutation_importance

plt.style.use('seaborn-v0_8-whitegrid')
os.makedirs('models', exist_ok=True)

print('Γ£à Library berhasil diimport')

--- CELL 2 (code) ---
# ============================================================
# CELL 2: Load Data & Feature Columns
# ============================================================
df_train = pd.read_csv('dataset/data_train.csv', parse_dates=['waktu_pembacaan'])
df_test  = pd.read_csv('dataset/data_test.csv',  parse_dates=['waktu_pembacaan'])

with open('models/feature_cols.json') as f:
    feature_cols = json.load(f)

TARGET = 'target_kelembaban_next'

X_train = df_train[feature_cols]
y_train = df_train[TARGET]
X_test  = df_test[feature_cols]
y_test  = df_test[TARGET]

print('=== ≡ƒôé Data Dimuat ===')
print(f'X_train : {X_train.shape}')
print(f'X_test  : {X_test.shape}')
print(f'Fitur   : {len(feature_cols)}')
print(f'Target  : {TARGET}')
print(f'y_train : mean={y_train.mean():.2f}%, std={y_train.std():.2f}%')
print(f'y_test  : mean={y_test.mean():.2f}%, std={y_test.std():.2f}%')

--- CELL 3 (code) ---
# ============================================================
# CELL 3: Training Random Forest ΓÇô Model Baseline
# ============================================================
print('=== ≡ƒî▓ Melatih Random Forest Regressor ===')
print('Parameter: n_estimators=200, max_depth=20, min_samples_leaf=5')
print('Mohon tunggu...')

rf_model = RandomForestRegressor(
    n_estimators   = 200,      # 200 pohon
    max_depth      = 20,       # kedalaman max pohon
    min_samples_leaf = 5,      # min sampel di leaf
    max_features   = 'sqrt',   # sqrt(n_features) per split
    n_jobs         = -1,       # pakai semua CPU core
    random_state   = 42,
    oob_score      = True      # Out-of-Bag score untuk validasi internal
)

rf_model.fit(X_train, y_train)

print(f'\nΓ£à Training selesai!')
print(f'OOB Score (R┬▓): {rf_model.oob_score_:.4f}')

--- CELL 4 (code) ---
# ============================================================
# CELL 4: Prediksi & Evaluasi Metrik
# ============================================================
y_pred_train = rf_model.predict(X_train)
y_pred_test  = rf_model.predict(X_test)

def evaluasi_model(y_true, y_pred, nama_set):
    mae  = mean_absolute_error(y_true, y_pred)
    rmse = np.sqrt(mean_squared_error(y_true, y_pred))
    r2   = r2_score(y_true, y_pred)
    mape = np.mean(np.abs((y_true - y_pred) / (y_true + 1e-8))) * 100
    print(f'--- {nama_set} ---')
    print(f'  MAE  : {mae:.4f}%')
    print(f'  RMSE : {rmse:.4f}%')
    print(f'  R┬▓   : {r2:.4f}')
    print(f'  MAPE : {mape:.4f}%')
    return {'MAE': mae, 'RMSE': rmse, 'R2': r2, 'MAPE': mape}

print('=== ≡ƒôÅ Evaluasi Model Random Forest ===')
rf_train_metrics = evaluasi_model(y_train, y_pred_train, 'Training')
print()
rf_test_metrics  = evaluasi_model(y_test,  y_pred_test,  'Testing')

# Simpan hasil metrik
rf_metrics = {'train': rf_train_metrics, 'test': rf_test_metrics}
with open('models/rf_metrics.json', 'w') as f:
    json.dump(rf_metrics, f, indent=2)
print('\nΓ£à Metrik disimpan ke models/rf_metrics.json')

axes[0].plot(waktu_plot, y_pred_test[-n_plot:],

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import json
import pickle
import os
import warnings
warnings.filterwarnings('ignore')

import xgboost as xgb
from xgboost import XGBRegressor
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score
from sklearn.model_selection import GridSearchCV, TimeSeriesSplit

plt.style.use('seaborn-v0_8-whitegrid')
os.makedirs('models', exist_ok=True)

print(f'Γ£à Library berhasil diimport')
print(f'   XGBoost version: {xgb.__version__}')

--- CELL 2 (code) ---
# ============================================================
# CELL 2: Load Data & Feature Columns
# ============================================================
df_train = pd.read_csv('dataset/data_train.csv', parse_dates=['waktu_pembacaan'])
df_test  = pd.read_csv('dataset/data_test.csv',  parse_dates=['waktu_pembacaan'])

with open('models/feature_cols.json') as f:
    feature_cols = json.load(f)

TARGET = 'target_kelembaban_next'

X_train = df_train[feature_cols]
y_train = df_train[TARGET]
X_test  = df_test[feature_cols]
y_test  = df_test[TARGET]

print('=== ≡ƒôé Data Dimuat ===')
print(f'X_train : {X_train.shape}')
print(f'X_test  : {X_test.shape}')
print(f'Target  : {TARGET}')

--- CELL 3 (code) ---
# ============================================================
# CELL 3: XGBoost ΓÇô Training Awal (Default Params)
# ============================================================
print('=== ΓÜí XGBoost ΓÇô Training Awal (Baseline) ===')

xgb_base = XGBRegressor(
    n_estimators    = 500,
    learning_rate   = 0.05,
    max_depth       = 8,
    subsample       = 0.8,
    colsample_bytree= 0.8,
    min_child_weight= 5,
    reg_alpha       = 0.1,
    reg_lambda      = 1.0,
    random_state    = 42,
    n_jobs          = -1,
    early_stopping_rounds = 30,
    eval_metric     = 'rmse'
)

xgb_base.fit(
    X_train, y_train,
    eval_set=[(X_train, y_train), (X_test, y_test)],
    verbose=50
)

best_iter = xgb_base.best_iteration
print(f'\nΓ£à Training selesai! Best iteration: {best_iter}')

--- CELL 4 (code) ---
# ============================================================
# CELL 4: Plot Learning Curve XGBoost
# ============================================================
results = xgb_base.evals_result()
train_rmse = results['validation_0']['rmse']
test_rmse  = results['validation_1']['rmse']

fig, ax = plt.subplots(figsize=(12, 5))
ax.plot(train_rmse, color='#1565C0', linewidth=1.5, label='Training RMSE')
ax.plot(test_rmse,  color='#E65100', linewidth=1.5, label='Testing RMSE')
ax.axvline(x=best_iter, color='green', linestyle='--', alpha=0.7, label=f'Best Iter ({best_iter})')
ax.set_xlabel('Boosting Round')
ax.set_ylabel('RMSE')
ax.set_title('ΓÜí XGBoost ΓÇô Learning Curve (RMSE)', fontsize=13, fontweight='bold')
ax.legend()
plt.tight_layout()
plt.savefig('dataset/plot_xgb_learning_curve.png', dpi=150, bbox_inches='tight')
plt.show()
print('Γ£à Learning curve disimpan')

--- CELL 5 (code) ---
# ============================================================
# CELL 5: Evaluasi XGBoost Baseline
# ============================================================
y_pred_train_base = xgb_base.predict(X_train)
y_pred_test_base  = xgb_base.predict(X_test)

def evaluasi_model(y_true, y_pred, nama_set):
    mae  = mean_absolute_error(y_true, y_pred)
    rmse = np.sqrt(mean_squared_error(y_true, y_pred))
    r2   = r2_score(y_true, y_pred)
    mape = np.mean(np.abs((y_true - y_pred) / (y_true + 1e-8))) * 100
    print(f'--- {nama_set} ---')
    print(f'  MAE  : {mae:.4f}%')
    print(f'  RMSE : {rmse:.4f}%')
    print(f'  R┬▓   : {r2:.4f}')
    print(f'  MAPE : {mape:.4f}%')
    return {'MAE': mae, 'RMSE': rmse, 'R2': r2, 'MAPE': mape}

print('=== ≡ƒôÅ Evaluasi XGBoost Baseline ===')
xgb_train_metrics = evaluasi_model(y_train, y_pred_train_base, 'Training')
print()
xgb_test_metrics  = evaluasi_model(y_test,  y_pred_test_base,  'Testing')

--- CELL 6 (code) ---
# ============================================================
# CELL 6: Hyperparameter Tuning ΓÇô GridSearchCV dengan TimeSeriesSplit
# ============================================================
print('=== ≡ƒöº Hyperparameter Tuning XGBoost ===')
print('Menggunakan TimeSeriesSplit (5-fold) untuk menghormati urutan waktu...')
print('Proses ini mungkin memakan waktu 5-15 menit...\n')

# TimeSeriesSplit untuk data time-series (TIDAK menggunakan shuffle)
tscv = TimeSeriesSplit(n_splits=5)

param_grid = {
    'n_estimators'    : [300, 500],
    'max_depth'       : [6, 8, 10],
    'learning_rate'   : [0.03, 0.05, 0.1],
    'subsample'       : [0.7, 0.8],
    'min_child_weight': [3, 5]
}

xgb_cv = XGBRegressor(
    colsample_bytree = 0.8,
    reg_alpha        = 0.1,
    reg_lambda       = 1.0,
    random_state     = 42,
    n_jobs           = -1,
    eval_metric      = 'rmse'
)

grid_search = GridSearchCV(
    estimator  = xgb_cv,
    param_grid = param_grid,
    cv         = tscv,
    scoring    = 'neg_root_mean_squared_error',
    n_jobs     = -1,
    verbose    = 1
)

grid_search.fit(X_train, y_train)

print(f'\nΓ£à Tuning selesai!')
print(f'Best RMSE (CV): {-grid_search.best_score_:.4f}')
print(f'Best Params   :')
for k, v in grid_search.best_params_.items():
    print(f'  {k}: {v}')

--- CELL 7 (code) ---
# ============================================================
# CELL 7: Training XGBoost dengan Parameter Terbaik
# ============================================================
print('=== ΓÜí Training XGBoost ΓÇô Parameter Terbaik ===')

best_params = grid_search.best_params_

xgb_best = XGBRegressor(
    **best_params,
    colsample_bytree      = 0.8,
    reg_alpha             = 0.1,
    reg_lambda            = 1.0,
    random_state          = 42,
    n_jobs                = -1,
    early_stopping_rounds = 30,
    eval_metric           = 'rmse'
)

xgb_best.fit(
    X_train, y_train,
    eval_set=[(X_train, y_train), (X_test, y_test)],
    verbose=50
)

print(f'\nΓ£à Training selesai! Best iteration: {xgb_best.best_iteration}')

--- CELL 9 (code) ---
# ============================================================
# CELL 8: Evaluasi Model Terbaik
# ============================================================
y_pred_train = xgb_best.predict(X_train)
y_pred_test  = xgb_best.predict(X_test)

print('=== ≡ƒôÅ Evaluasi XGBoost ΓÇô Parameter Terbaik ===')
xgb_train_metrics_best = evaluasi_model(y_train, y_pred_train, 'Training (Best)')
print()
xgb_test_metrics_best  = evaluasi_model(y_test,  y_pred_test,  'Testing (Best)')

# Simpan metrik
xgb_metrics = {

with open('models/xgb_metrics.json', 'w') as f:
